## 概述

智能体系统在实现代码生成时，需要解决两个核心问题：
1. **代码规划**：如何将复杂任务分解为可执行的代码模块
2. **代码协同**：如何确保各模块之间变量定义正确、接口衔接自然

本文将深入到代码实现层面，详细讲解这两个机制的实现细节。

## 代码规划机制

### 2.1 规划架构

智能体的代码规划基于以下核心组件：

```python
class CodePlanner:
    """代码规划器：负责将任务分解为可执行的代码计划"""
    
    def __init__(self):
        self.task_graph = {}  # 任务依赖图
        self.variable_registry = {}  # 变量注册表
        self.module_dependencies = []  # 模块依赖列表
        self.execution_order = []  # 执行顺序
        
    def analyze_task(self, task_description):
        """
        分析任务需求，提取关键信息
        
        Args:
            task_description: 任务描述文本
            
        Returns:
            task_structure: 结构化的任务信息
        """
        task_structure = {
            'main_goal': None,
            'sub_tasks': [],
            'required_inputs': [],
            'expected_outputs': [],
            'constraints': []
        }
        
        # 1. 使用NLP技术提取任务目标
        task_structure['main_goal'] = self._extract_main_goal(task_description)
        
        # 2. 识别子任务
        task_structure['sub_tasks'] = self._identify_subtasks(task_description)
        
        # 3. 分析输入输出需求
        task_structure['required_inputs'] = self._extract_inputs(task_description)
        task_structure['expected_outputs'] = self._extract_outputs(task_description)
        
        # 4. 识别约束条件
        task_structure['constraints'] = self._extract_constraints(task_description)
        
        return task_structure
    
    def create_execution_plan(self, task_structure):
        """
        创建执行计划，确定任务执行顺序
        
        Args:
            task_structure: 结构化任务信息
            
        Returns:
            execution_plan: 执行计划对象
        """
        execution_plan = {
            'steps': [],
            'dependencies': {},
            'variable_flow': {}
        }
        
        # 1. 构建任务依赖图
        dependency_graph = self._build_dependency_graph(task_structure['sub_tasks'])
        
        # 2. 拓扑排序确定执行顺序
        execution_order = self._topological_sort(dependency_graph)
        
        # 3. 为每个步骤创建详细计划
        for task_id in execution_order:
            step = self._create_step_plan(
                task_id, 
                task_structure['sub_tasks'][task_id],
                dependency_graph
            )
            execution_plan['steps'].append(step)
            
        # 4. 分析变量流转
        execution_plan['variable_flow'] = self._analyze_variable_flow(
            execution_plan['steps']
        )
        
        return execution_plan
    
    def _build_dependency_graph(self, sub_tasks):
        """
        构建任务依赖图
        
        实现细节：
        1. 分析每个子任务的输入输出
        2. 识别任务间的数据依赖关系
        3. 构建有向无环图(DAG)
        """
        graph = {}
        
        for i, task in enumerate(sub_tasks):
            graph[i] = {
                'task': task,
                'inputs': self._extract_task_inputs(task),
                'outputs': self._extract_task_outputs(task),
                'dependencies': []
            }
        
        # 建立依赖关系
        for i in range(len(sub_tasks)):
            for j in range(len(sub_tasks)):
                if i != j:
                    # 如果任务i的输出是任务j的输入，则j依赖于i
                    if self._has_data_dependency(graph[i], graph[j]):
                        graph[j]['dependencies'].append(i)
        
        return graph
    
    def _topological_sort(self, graph):
        """
        拓扑排序算法实现
        
        使用Kahn算法确保任务按正确顺序执行
        """
        # 计算每个节点的入度
        in_degree = {node: 0 for node in graph}
        for node in graph:
            for dep in graph[node]['dependencies']:
                in_degree[node] += 1
        
        # 找到所有入度为0的节点
        queue = [node for node in in_degree if in_degree[node] == 0]
        result = []
        
        while queue:
            # 取出入度为0的节点
            node = queue.pop(0)
            result.append(node)
            
            # 更新依赖该节点的其他节点的入度
            for next_node in graph:
                if node in graph[next_node]['dependencies']:
                    in_degree[next_node] -= 1
                    if in_degree[next_node] == 0:
                        queue.append(next_node)
        
        # 检查是否存在循环依赖
        if len(result) != len(graph):
            raise ValueError("检测到循环依赖，无法生成执行计划")
        
        return result
```

```python
class TaskDecomposer:
    """任务分解器：将复杂任务分解为简单子任务"""
    
    def decompose(self, complex_task):
        """
        任务分解的核心算法
        
        分解策略：
        1. 功能分解：按功能模块拆分
        2. 数据分解：按数据流程拆分
        3. 时序分解：按执行顺序拆分
        """
        sub_tasks = []
        
        # 策略1：识别独立的功能模块
        functional_modules = self._identify_functional_modules(complex_task)
        
        # 策略2：分析数据处理流程
        data_pipeline = self._analyze_data_pipeline(complex_task)
        
        # 策略3：确定执行阶段
        execution_phases = self._determine_execution_phases(complex_task)
        
        # 综合三种策略生成子任务
        sub_tasks = self._synthesize_subtasks(
            functional_modules,
            data_pipeline,
            execution_phases
        )
        
        return sub_tasks
    
    def _identify_functional_modules(self, task):
        """
        识别功能模块
        
        实现细节：
        - 使用关键词匹配识别常见模块（如：数据加载、预处理、模型训练等）
        - 分析任务描述中的动词短语
        - 识别输入输出关系
        """
        modules = []
        
        # 定义常见功能模块的特征
        module_patterns = {
            'data_loading': ['加载', '读取', '导入', 'load', 'read'],
            'preprocessing': ['预处理', '清洗', '转换', 'preprocess', 'clean'],
            'processing': ['处理', '计算', '分析', 'process', 'compute'],
            'output': ['保存', '输出', '导出', 'save', 'export']
        }
        
        for module_type, keywords in module_patterns.items():
            if any(keyword in task for keyword in keywords):
                modules.append({
                    'type': module_type,
                    'description': task,
                    'priority': self._calculate_priority(module_type)
                })
        
        return modules
```

### 3.1 变量注册表系统

这是确保代码协同的核心机制：

```python
class VariableRegistry:
    """
    变量注册表：跟踪所有变量的定义和使用
    
    核心功能：
    1. 记录变量定义位置
    2. 追踪变量作用域
    3. 检测未定义变量
    4. 管理变量生命周期
    """
    
    def __init__(self):
        self.variables = {}  # 变量字典
        self.scopes = []  # 作用域栈
        self.current_scope = 0  # 当前作用域ID
        
    def register_variable(self, var_name, var_type, scope, definition_location):
        """
        注册新变量
        
        Args:
            var_name: 变量名
            var_type: 变量类型
            scope: 作用域ID
            definition_location: 定义位置（模块、行号）
        """
        if var_name not in self.variables:
            self.variables[var_name] = []
        
        var_info = {
            'name': var_name,
            'type': var_type,
            'scope': scope,
            'defined_at': definition_location,
            'used_by': [],  # 记录哪些模块使用了这个变量
            'dependencies': [],  # 记录这个变量依赖哪些其他变量
            'status': 'defined'  # 状态：defined, used, modified
        }
        
        self.variables[var_name].append(var_info)
        
    def check_variable_defined(self, var_name, usage_location):
        """
        检查变量是否已定义
        
        Returns:
            (bool, var_info): 是否定义，变量信息
        """
        current_scope = self._get_current_scope()
        
        # 在当前作用域及父作用域中查找
        for scope in self._get_scope_chain(current_scope):
            for var_info in self.variables.get(var_name, []):
                if var_info['scope'] == scope:
                    # 记录使用位置
                    var_info['used_by'].append(usage_location)
                    return True, var_info
        
        # 未找到定义
        return False, None
    
    def analyze_dependencies(self, code_block):
        """
        分析代码块的变量依赖
        
        返回：
        - required_variables: 需要预先定义的变量
        - produced_variables: 代码块产生的变量
        - variable_flow: 变量流转关系
        """
        import ast
        
        tree = ast.parse(code_block)
        
        required_variables = set()
        produced_variables = set()
        
        class VariableAnalyzer(ast.NodeVisitor):
            def __init__(self, registry):
                self.registry = registry
                self.current_scope_vars = set()
                
            def visit_Name(self, node):
                if isinstance(node.ctx, ast.Store):
                    # 变量被赋值（定义）
                    produced_variables.add(node.id)
                    self.current_scope_vars.add(node.id)
                elif isinstance(node.ctx, ast.Load):
                    # 变量被使用
                    if node.id not in self.current_scope_vars:
                        required_variables.add(node.id)
                        
            def visit_FunctionDef(self, node):
                # 函数参数也是定义
                for arg in node.args.args:
                    produced_variables.add(arg.arg)
                self.generic_visit(node)
        
        analyzer = VariableAnalyzer(self)
        analyzer.visit(tree)
        
        return {
            'required': list(required_variables),
            'produced': list(produced_variables),
            'flow': self._build_flow_graph(required_variables, produced_variables)
        }
```

### 3.2 依赖解析器

```python
class DependencyResolver:
    """
    依赖解析器：解决变量依赖问题
    
    主要功能：
    1. 识别缺失的变量定义
    2. 自动插入变量初始化代码
    3. 调整代码执行顺序
    """
    
    def __init__(self, variable_registry):
        self.registry = variable_registry
        
    def resolve_dependencies(self, code_modules):
        """
        解析并解决所有模块间的依赖关系
        
        Args:
            code_modules: 代码模块列表
            
        Returns:
            resolved_modules: 解决依赖后的模块列表（可能包含新增的初始化模块）
        """
        # 1. 分析每个模块的依赖
        module_dependencies = []
        for i, module in enumerate(code_modules):
            deps = self.registry.analyze_dependencies(module['code'])
            module_dependencies.append({
                'module_id': i,
                'module': module,
                'required': deps['required'],
                'produced': deps['produced']
            })
        
        # 2. 检测未定义的变量
        undefined_vars = self._find_undefined_variables(module_dependencies)
        
        # 3. 生成变量初始化代码
        init_modules = []
        if undefined_vars:
            init_modules = self._generate_initialization_modules(undefined_vars)
        
        # 4. 重新排序模块
        ordered_modules = self._reorder_modules(
            init_modules + module_dependencies
        )
        
        # 5. 验证依赖已解决
        self._verify_dependencies(ordered_modules)
        
        return [m['module'] for m in ordered_modules]
    
    def _find_undefined_variables(self, module_dependencies):
        """
        查找未定义的变量
        
        实现逻辑：
        1. 遍历所有模块
        2. 检查每个模块需要的变量是否在之前的模块中被定义
        3. 收集所有未定义的变量
        """
        undefined = set()
        defined = set()
        
        for dep in module_dependencies:
            # 检查需要的变量是否已定义
            for var in dep['required']:
                if var not in defined:
                    undefined.add(var)
            
            # 更新已定义变量集合
            defined.update(dep['produced'])
        
        return undefined
    
    def _generate_initialization_modules(self, undefined_vars):
        """
        为未定义的变量生成初始化代码
        
        策略：
        1. 根据变量名推测类型
        2. 生成合适的初始化代码
        3. 添加注释说明
        """
        init_modules = []
        
        for var in undefined_vars:
            var_type = self._infer_variable_type(var)
            init_code = self._generate_init_code(var, var_type)
            
            init_modules.append({
                'module_id': f'init_{var}',
                'module': {
                    'code': init_code,
                    'type': 'initialization',
                    'description': f'初始化变量 {var}'
                },
                'required': [],
                'produced': [var]
            })
        
        return init_modules
    
    def _infer_variable_type(self, var_name):
        """
        根据变量名推测类型
        
        启发式规则：
        - 以_list结尾 -> list
        - 以_dict结尾 -> dict
        - 以_df结尾 -> DataFrame
        - 以num_, count_开头 -> int
        - 以is_, has_开头 -> bool
        - 默认 -> None (需要用户指定)
        """
        if var_name.endswith('_list'):
            return 'list'
        elif var_name.endswith('_dict'):
            return 'dict'
        elif var_name.endswith('_df'):
            return 'DataFrame'
        elif var_name.startswith(('num_', 'count_')):
            return 'int'
        elif var_name.startswith(('is_', 'has_')):
            return 'bool'
        else:
            return 'unknown'
    
    def _generate_init_code(self, var_name, var_type):
        """
        生成初始化代码
        """
        init_templates = {
            'list': f'{var_name} = []',
            'dict': f'{var_name} = {{}}',
            'DataFrame': f'{var_name} = pd.DataFrame()',
            'int': f'{var_name} = 0',
            'bool': f'{var_name} = False',
            'unknown': f'{var_name} = None  # TODO: 请指定初始值'
        }
        
        code = f"# 自动生成的变量初始化\n"
        code += init_templates.get(var_type, init_templates['unknown'])
        
        return code
```

```python
class InterfaceManager:
    """
    接口管理器：确保模块间接口一致
    
    功能：
    1. 定义模块接口规范
    2. 验证接口兼容性
    3. 生成接口适配代码
    """
    
    def __init__(self):
        self.interfaces = {}
        
    def define_interface(self, module_id, interface_spec):
        """
        定义模块接口
        
        Args:
            module_id: 模块标识
            interface_spec: 接口规范
                {
                    'inputs': [{'name': 'x', 'type': 'int', 'description': '...'}],
                    'outputs': [{'name': 'result', 'type': 'float', 'description': '...'}],
                    'side_effects': ['修改全局变量X']
                }
        """
        self.interfaces[module_id] = interface_spec
        
    def check_compatibility(self, producer_id, consumer_id):
        """
        检查两个模块的接口是否兼容
        
        Returns:
            (bool, issues): 是否兼容，不兼容的原因列表
        """
        producer = self.interfaces[producer_id]
        consumer = self.interfaces[consumer_id]
        
        issues = []
        
        # 检查生产者的输出是否满足消费者的输入
        producer_outputs = {o['name']: o for o in producer['outputs']}
        
        for input_spec in consumer['inputs']:
            if input_spec['name'] not in producer_outputs:
                issues.append(f"缺少输入: {input_spec['name']}")
            else:
                output_spec = producer_outputs[input_spec['name']]
                if not self._type_compatible(output_spec['type'], input_spec['type']):
                    issues.append(
                        f"类型不匹配: {input_spec['name']} "
                        f"需要 {input_spec['type']}, "
                        f"得到 {output_spec['type']}"
                    )
        
        return len(issues) == 0, issues
    
    def generate_adapter(self, producer_id, consumer_id):
        """
        生成适配器代码，使两个模块能够协同工作
        
        返回适配器代码字符串
        """
        compatible, issues = self.check_compatibility(producer_id, consumer_id)
        
        if compatible:
            return ""  # 无需适配器
        
        adapter_code = "# 自动生成的适配器代码\n\n"
        
        producer = self.interfaces[producer_id]
        consumer = self.interfaces[consumer_id]
        
        producer_outputs = {o['name']: o for o in producer['outputs']}
        
        for input_spec in consumer['inputs']:
            if input_spec['name'] not in producer_outputs:
                # 生成默认值
                adapter_code += self._generate_default_value(input_spec)
            else:
                output_spec = producer_outputs[input_spec['name']]
                if not self._type_compatible(output_spec['type'], input_spec['type']):
                    # 生成类型转换代码
                    adapter_code += self._generate_type_conversion(
                        input_spec['name'],
                        output_spec['type'],
                        input_spec['type']
                    )
        
        return adapter_code
    
    def _type_compatible(self, type1, type2):
        """检查类型是否兼容"""
        # 简化的类型兼容性检查
        if type1 == type2:
            return True
        
        # 数值类型之间的兼容
        numeric_types = {'int', 'float', 'number'}
        if type1 in numeric_types and type2 in numeric_types:
            return True
        
        return False
    
    def _generate_type_conversion(self, var_name, from_type, to_type):
        """生成类型转换代码"""
        conversion_map = {
            ('int', 'float'): f'{var_name} = float({var_name})',
            ('float', 'int'): f'{var_name} = int({var_name})',
            ('str', 'int'): f'{var_name} = int({var_name})',
            ('int', 'str'): f'{var_name} = str({var_name})',
        }
        
        key = (from_type, to_type)
        if key in conversion_map:
            return conversion_map[key] + '\n'
        else:
            return f'# TODO: 需要手动转换 {var_name} 从 {from_type} 到 {to_type}\n'
```

### 4.2 代码生成协调器

```python
class CodeGenerationCoordinator:
    """
    代码生成协调器：统筹整个代码生成过程
    
    这是将所有组件整合在一起的核心类
    """
    
    def __init__(self):
        self.planner = CodePlanner()
        self.registry = VariableRegistry()
        self.resolver = DependencyResolver(self.registry)
        self.interface_mgr = InterfaceManager()
        
    def generate_code(self, task_description):
        """
        完整的代码生成流程
        
        Args:
            task_description: 任务描述
            
        Returns:
            final_code: 最终生成的完整代码
        """
        # 第1阶段：规划
        print("第1阶段：任务分析与规划...")
        task_structure = self.planner.analyze_task(task_description)
        execution_plan = self.planner.create_execution_plan(task_structure)
        
        # 第2阶段：生成初始代码模块
        print("第2阶段：生成代码模块...")
        code_modules = []
        for step in execution_plan['steps']:
            module = self._generate_module_code(step)
            code_modules.append(module)
            
            # 注册模块中的变量
            self._register_module_variables(module)
        
        # 第3阶段：解决依赖关系
        print("第3阶段：解决依赖关系...")
        resolved_modules = self.resolver.resolve_dependencies(code_modules)
        
        # 第4阶段：生成接口适配代码
        print("第4阶段：生成接口适配代码...")
        modules_with_adapters = []
        for i in range(len(resolved_modules)):
            modules_with_adapters.append(resolved_modules[i])
            
            # 在相邻模块之间插入适配器
            if i < len(resolved_modules) - 1:
                adapter = self.interface_mgr.generate_adapter(
                    resolved_modules[i]['id'],
                    resolved_modules[i + 1]['id']
                )
                if adapter:
                    modules_with_adapters.append({
                        'code': adapter,
                        'type': 'adapter'
                    })
        
        # 第5阶段：组装最终代码
        print("第5阶段：组装最终代码...")
        final_code = self._assemble_final_code(modules_with_adapters)
        
        # 第6阶段：验证
        print("第6阶段：验证代码...")
        self._validate_code(final_code)
        
        return final_code
    
    def _generate_module_code(self, step):
        """
        根据执行步骤生成具体的代码模块
        
        这里需要结合LLM或代码模板来生成
        """
        module = {
            'id': step['id'],
            'code': '',
            'type': step['type'],
            'description': step['description']
        }
        
        # 根据步骤类型选择代码模板或生成策略
        if step['type'] == 'data_loading':
            module['code'] = self._generate_data_loading_code(step)
        elif step['type'] == 'preprocessing':
            module['code'] = self._generate_preprocessing_code(step)
        elif step['type'] == 'processing':
            module['code'] = self._generate_processing_code(step)
        elif step['type'] == 'output':
            module['code'] = self._generate_output_code(step)
        
        return module
    
    def _register_module_variables(self, module):
        """注册模块中的所有变量"""
        deps = self.registry.analyze_dependencies(module['code'])
        
        for var in deps['produced']:
            self.registry.register_variable(
                var_name=var,
                var_type='unknown',  # 需要进一步推断
                scope=module['id'],
                definition_location=module['id']
            )
    
    def _assemble_final_code(self, modules):
        """
        组装最终代码
        
        包括：
        1. 添加必要的导入语句
        2. 添加文档字符串
        3. 组合所有模块
        4. 格式化代码
        """
        code_parts = []
        
        # 1. 收集所有需要的导入
        imports = self._collect_imports(modules)
        if imports:
            code_parts.append("# 导入必要的库")
            code_parts.extend(imports)
            code_parts.append("")
        
        # 2. 添加主文档
        code_parts.append('"""')
        code_parts.append("自动生成的代码")
        code_parts.append("本代码由智能体系统自动规划和生成")
        code_parts.append('"""')
        code_parts.append("")
        
        # 3. 添加所有模块
        for i, module in enumerate(modules):
            # 添加模块注释
            if 'description' in module:
                code_parts.append(f"# 模块 {i + 1}: {module['description']}")
            
            # 添加模块代码
            code_parts.append(module['code'])
            code_parts.append("")  # 模块间空行
        
        # 4. 组合并格式化
        final_code = '\n'.join(code_parts)
        
        # 使用代码格式化工具（如black或autopep8）
        try:
            import black
            final_code = black.format_str(final_code, mode=black.Mode())
        except:
            pass  # 如果格式化失败，使用原始代码
        
        return final_code
    
    def _validate_code(self, code):
        """
        验证生成的代码
        
        检查：
        1. 语法正确性
        2. 所有变量都已定义
        3. 导入语句完整
        """
        import ast
        
        try:
            # 语法检查
            ast.parse(code)
            print("✓ 语法检查通过")
        except SyntaxError as e:
            print(f"✗ 语法错误: {e}")
            raise
        
        # 变量定义检查
        deps = self.registry.analyze_dependencies(code)
        undefined = [v for v in deps['required'] if not self._is_builtin(v)]
        
        if undefined:
            print(f"⚠ 警告：以下变量可能未定义: {undefined}")
        else:
            print("✓ 变量定义检查通过")
    
    def _is_builtin(self, var_name):
        """检查是否是内置变量或函数"""
        import builtins
        return hasattr(builtins, var_name)
```

```python
#!/usr/bin/env python3
"""
智能体代码规划与协同实现 - 完整示例

这是一个完整的、可运行的示例，展示了智能体系统如何进行代码规划和协同。
"""

import ast
import re
from typing import List, Dict, Any, Set, Tuple


class VariableRegistry:
    """变量注册表"""
    
    def __init__(self):
        self.variables = {}
        self.scopes = []
        
    def analyze_dependencies(self, code: str) -> Dict[str, List[str]]:
        """分析代码的变量依赖"""
        try:
            tree = ast.parse(code)
        except SyntaxError:
            return {'required': [], 'produced': []}
        
        required = set()
        produced = set()
        
        class VariableAnalyzer(ast.NodeVisitor):
            def __init__(self):
                self.local_vars = set()
                
            def visit_Name(self, node):
                if isinstance(node.ctx, ast.Store):
                    produced.add(node.id)
                    self.local_vars.add(node.id)
                elif isinstance(node.ctx, ast.Load):
                    if node.id not in self.local_vars:
                        required.add(node.id)
                        
            def visit_FunctionDef(self, node):
                produced.add(node.name)
                for arg in node.args.args:
                    self.local_vars.add(arg.arg)
                self.generic_visit(node)
                
            def visit_Import(self, node):
                for alias in node.names:
                    name = alias.asname if alias.asname else alias.name
                    produced.add(name)
                    
            def visit_ImportFrom(self, node):
                for alias in node.names:
                    name = alias.asname if alias.asname else alias.name
                    produced.add(name)
        
        analyzer = VariableAnalyzer()
        analyzer.visit(tree)
        
        # 过滤内置函数和常见库
        builtins = {'print', 'len', 'range', 'str', 'int', 'float', 'list', 'dict', 
                   'set', 'tuple', 'True', 'False', 'None'}
        required = required - builtins - produced
        
        return {
            'required': sorted(list(required)),
            'produced': sorted(list(produced))
        }


class DependencyResolver:
    """依赖解析器"""
    
    def __init__(self, registry: VariableRegistry):
        self.registry = registry
        
    def resolve_dependencies(self, modules: List[Dict]) -> List[Dict]:
        """解析并解决模块间的依赖关系"""
        # 分析每个模块的依赖
        module_deps = []
        for i, module in enumerate(modules):
            deps = self.registry.analyze_dependencies(module['code'])
            module_deps.append({
                'index': i,
                'module': module,
                'required': set(deps['required']),
                'produced': set(deps['produced'])
            })
        
        # 找出未定义的变量
        undefined = self._find_undefined_variables(module_deps)
        
        # 生成初始化模块
        init_modules = []
        if undefined:
            for var in sorted(undefined):
                init_code = self._generate_init_code(var)
                init_modules.append({
                    'index': -1,
                    'module': {
                        'code': init_code,
                        'description': f'初始化变量 {var}'
                    },
                    'required': set(),
                    'produced': {var}
                })
        
        # 重新排序模块
        all_modules = init_modules + module_deps
        ordered = self._topological_sort(all_modules)
        
        return [m['module'] for m in ordered]
    
    def _find_undefined_variables(self, module_deps: List[Dict]) -> Set[str]:
        """找出未定义的变量"""
        undefined = set()
        defined = set()
        
        for dep in module_deps:
            for var in dep['required']:
                if var not in defined:
                    undefined.add(var)
            defined.update(dep['produced'])
        
        return undefined
    
    def _generate_init_code(self, var_name: str) -> str:
        """生成初始化代码"""
        var_type = self._infer_type(var_name)
        
        templates = {
            'list': f'{var_name} = []',
            'dict': f'{var_name} = {{}}',
            'int': f'{var_name} = 0',
            'float': f'{var_name} = 0.0',
            'str': f'{var_name} = ""',
            'bool': f'{var_name} = False',
            'DataFrame': f'{var_name} = pd.DataFrame()',
            'unknown': f'{var_name} = None  # TODO: 请指定初始值'
        }
        
        init_code = f"# 自动生成的变量初始化\n"
        init_code += templates.get(var_type, templates['unknown'])
        
        return init_code
    
    def _infer_type(self, var_name: str) -> str:
        """根据变量名推测类型"""
        if var_name.endswith('_list') or var_name.endswith('s'):
            return 'list'
        elif var_name.endswith('_dict'):
            return 'dict'
        elif var_name.endswith('_df') or var_name.startswith('df_'):
            return 'DataFrame'
        elif var_name.startswith(('num_', 'count_', 'n_')):
            return 'int'
        elif var_name.startswith(('is_', 'has_', 'should_')):
            return 'bool'
        elif var_name.endswith('_str') or var_name.startswith('str_'):
            return 'str'
        else:
            return 'unknown'
    
    def _topological_sort(self, modules: List[Dict]) -> List[Dict]:
        """拓扑排序"""
        # 构建依赖图
        n = len(modules)
        graph = [[] for _ in range(n)]
        in_degree = [0] * n
        
        for i in range(n):
            for j in range(n):
                if i != j:
                    # 如果模块j需要模块i产生的变量
                    if modules[j]['required'] & modules[i]['produced']:
                        graph[i].append(j)
                        in_degree[j] += 1
        
        # Kahn算法
        queue = [i for i in range(n) if in_degree[i] == 0]
        result = []
        
        while queue:
            node = queue.pop(0)
            result.append(modules[node])
            
            for next_node in graph[node]:
                in_degree[next_node] -= 1
                if in_degree[next_node] == 0:
                    queue.append(next_node)
        
        if len(result) != n:
            # 存在循环依赖，返回原始顺序
            return modules
        
        return result
```
